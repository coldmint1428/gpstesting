-- How long each server tick took during one stress run. Paste into Supabase -> SQL Editor and press Run.
-- (cron.job_run_details can only be read here, not with the app's public key.) The ST1 results file has this
-- with the run's times already filled in; by hand, replace 2026-10-26T01:59:49.281Z and 2026-10-26T02:09:49.898Z with the run's start and end.
-- Pass for ST1: gps-move-tick p95 under 2 s (it runs every 5 s), nothing failed.
select j.jobname,
       count(*)                                                                         as runs,
       count(*) filter (where d.status <> 'succeeded')                                   as failed,
       round(percentile_cont(0.5) within group (order by extract(epoch from d.end_time - d.start_time))::numeric, 3) as median_s,
       round(percentile_cont(0.95) within group (order by extract(epoch from d.end_time - d.start_time))::numeric, 3) as p95_s,
       round(max(extract(epoch from d.end_time - d.start_time))::numeric, 3)             as max_s,
       min(d.return_message) filter (where d.status <> 'succeeded')                      as first_error
from cron.job_run_details d
join cron.job j on j.jobid = d.jobid
where j.jobname in ('gps-sim-tick', 'gps-move-tick', 'gps-show-tick', 'gps-push-send', 'gps-route-eta')
  and d.start_time between '2026-10-26T01:59:49.281Z'::timestamptz and '2026-10-26T02:09:49.898Z'::timestamptz
group by j.jobname
order by j.jobname;
